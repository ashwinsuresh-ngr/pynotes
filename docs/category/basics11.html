class Calculator:

Overloading In-Place Operators
Wed 15 April 2026 
class Counter:
    def __init__(self, value):
        self.value = value

    def __iadd__(self, other):
        self.value += other
        return self

c = Counter(10)
c += 5
print(c.value)  # Output: 15
